        [UnityTest]
        public IEnumerator Board_CancelDeviceRemovalAndStaleBottleCannotCommitPendingRequests()
        {
            yield return _fixture.EnterMain();
            GameMenuController menu = Object.FindAnyObjectByType<GameMenuController>();
            _architecture = menu.GetArchitecture();
            var inventory = _architecture.GetModel<InventoryModel>();
            ItemBaseDefinition definition = ApplicationHost.Current.ContentCatalog.GetAll<ItemBaseDefinition>().Single(i => i.Id == "energy_bottle_small");
            ItemInstance bottle = definition.CreateInstance(_architecture.GetUtility<IItemInstanceIdGenerator>().Next(), 1, 42, ItemRarity.Normal);
            bottle.CollectEnergy(120); Assert.That(inventory.TryAddItem(bottle), Is.True);
            menu.OpenPage(GameMenuPage.SkillBoard); yield return null; yield return null;
            VisualElement root = menu.GetComponent<RuntimePanelView>().Root;
            Keyboard keyboard = InputSystem.AddDevice<Keyboard>();
            Gamepad pad = InputSystem.AddDevice<Gamepad>();
            FindItemButton(root, bottle, "inventory-item").Focus(); yield return null;
            yield return BoardPress(root, "board-choose-bottle", keyboard);
            Assert.That(menu.Board.Preview.Success, Is.True);
            _inputFixture.PressAndRelease(pad.buttonEast); yield return null; yield return null;
            Assert.That(menu.Board.Preview, Is.Null);
            Assert.That(inventory.Grid.Placements.ContainsKey(bottle), Is.True);
            yield return BoardPress(root, "board-choose-bottle", keyboard);
            InputSystem.RemoveDevice(pad); yield return null;
            Assert.That(menu.Board.Preview, Is.Null);
            yield return BoardPress(root, "board-choose-bottle", keyboard);
            Assert.That(inventory.RemoveItem(bottle), Is.True);
            Assert.That(root.Q<Button>("board-confirm").enabledInHierarchy, Is.False);
            Assert.That(_architecture.SendQuery(new GetPlayerGrowthQuery()).EarnedPoints, Is.Zero);
            Assert.That(menu.Board.Preview, Is.Null);
            Mouse mouse = InputSystem.AddDevice<Mouse>();
            _inputFixture.Set(mouse.position, PanelToScreen(root, menu.Board.Canvas.worldBound.center)); yield return null;
            _inputFixture.Press(mouse.middleButton); yield return null;
            Assert.That(menu.Board.Canvas.HasPointerCapture(PointerId.mousePointerId), Is.True, "中键平移须捕获指针");
            menu.ClosePage(GameMenuPage.SkillBoard); yield return null;
            Assert.That(menu.IsOpen, Is.False);
            Assert.That(menu.Board.Canvas.HasPointerCapture(PointerId.mousePointerId), Is.False, "关闭棋盘须释放平移捕获");
            _inputFixture.Release(mouse.middleButton); yield return null;
        }

        [UnityTest]
        public IEnumerator Board_LocalesKeepCanvasInventoryAndConfirmationReachable()
        {
            yield return _fixture.EnterMain();
            ApplicationHost host = ApplicationHost.Current;
            GameMenuController menu = Object.FindAnyObjectByType<GameMenuController>();
            VisualElement root = menu.GetComponent<RuntimePanelView>().Root;
            IArchitecture architecture = menu.GetArchitecture();
            PlayerController player = Object.FindAnyObjectByType<PlayerController>();
            architecture.GetSystem<PlayerGrowthSystem>().Restore(player.Id.Value, new PlayerGrowthState().Charge(1000), player.Actor);
            // 与正式 Restore 一致，派生效果重建后恢复绝对资源。
            player.Actor.RestoreResources(new CombatResourceSnapshot(player.Actor.MaxHealth, player.Actor.MaxHealth, player.Actor.MaxMana, player.Actor.MaxMana), true);
            foreach (int slot in new[] { 0, 2, 5 })
            {
                ItemInstance Create(string id)
                {
                    return host.ContentCatalog.GetAll<ItemBaseDefinition>().Single(d => d.Id == id).CreateInstance(architecture.GetUtility<IItemInstanceIdGenerator>().Next(), 1, slot, ItemRarity.Normal);
                }
                ItemInstance wood = Create("firewood_large"), flame = Create("flame_projectile");
                Assert.That(architecture.GetModel<InventoryModel>().TryAddItem(wood), Is.True);
                Assert.That(architecture.GetModel<InventoryModel>().TryAddItem(flame), Is.True);
                foreach (SkillBoardRequest request in new[] { new SkillBoardRequest(SkillBoardOperation.Install, wood, SkillBoardGeometry.Roots[slot]),
                    new SkillBoardRequest(SkillBoardOperation.Embed, flame, slot: new FirewoodPortReference(wood.InstanceId, "center")) })
                {
                    Assert.That(architecture.SendCommand(new CommitSkillBoardCommand(architecture.SendQuery(new PreviewSkillBoardQuery(player.Id.Value, request)))), Is.True);
                }
                Assert.That(architecture.SendCommand(new CommitPlayerGrowthCommand(architecture.SendQuery(new PreviewPlayerGrowthQuery(player.Id.Value, woodId: wood.InstanceId)))), Is.True);
                Assert.That(architecture.SendCommand(new AssignPlayerSkillCommand(player.Id.Value, slot, flame.InstanceId)), Is.True);
            }
            Vector2Int previousResolution = new Vector2Int(Screen.width, Screen.height);
            UserLanguagePreference previousLanguage = host.Settings.Current.Language;
            System.Type utility = System.Type.GetType("DarkFlare.Editor.Phase0GameViewResolutionUtility, DarkFlare.Editor", true);
            System.Reflection.MethodInfo setResolution = utility.GetMethod("SetResolution");
            try
            {
                setResolution.Invoke(null, new object[] { 1920, 1080 });
                yield return null; yield return null;
                menu.OpenPage(GameMenuPage.SkillBoard);
                foreach (string locale in new[] { "zh-Hans", "en", "qps-ploc" })
                {
                    yield return host.Localization.ChangeLanguageAsync(locale == "zh-Hans" ? UserLanguagePreference.SimplifiedChinese : UserLanguagePreference.English).ToCoroutine();
                    menu.Board.Refresh(); yield return null; yield return null;
                    AssertLayoutInsideRoot(root, new[] { "board-window", "inventory-window", "board-canvas", "board-confirm", "board-cancel", "board-skill-slots" });
                    AssertElementsDoNotOverlap(root, "board-window", "inventory-window");
                    Assert.That(root.Q("board-canvas").worldBound.height, Is.GreaterThan(100), locale);
                    Keyboard keyboard = InputSystem.AddDevice<Keyboard>();
                    yield return BoardPress(root, "board-root-5", keyboard);
                    Assert.That(menu.Board.Canvas.Cursor, Is.EqualTo(SkillBoardGeometry.Roots[5]));
                    yield return BoardPress(root, "board-center", keyboard);
                    Assert.That(menu.Board.Canvas.Cursor, Is.EqualTo(default(HexCoordinate)));
                    InputSystem.RemoveDevice(keyboard); yield return null; yield return null;
                    if (locale == "qps-ploc")
                    {
                        Assert.That(ApplyPseudoLocalization(root), Is.GreaterThan(0));
                        yield return null; yield return null;
                        AssertLayoutInsideRoot(root, new[] { "board-canvas", "board-confirm", "board-skill-slots" });
                    }
                    AssertVisibleTextFits(root, locale, new Vector2Int(1920, 1080));
                    // TEMP_FINAL_CAPTURE
                    if (locale == "qps-ploc")
                    {
                        ((LifecycleScope)typeof(SkillBoardPanelController).GetField("_refresh", System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic).GetValue(menu.Board)).BeginStop();
                    }
                    yield return new WaitForEndOfFrame();
                    Texture2D shot = ScreenCapture.CaptureScreenshotAsTexture();
                    System.IO.Directory.CreateDirectory("Temp/phase5-screenshots");
                    System.IO.File.WriteAllBytes("Temp/phase5-screenshots/board-" + locale + ".png", shot.EncodeToPNG());
                    Object.Destroy(shot);
                    // END_TEMP_FINAL_CAPTURE
                }
            }
            finally
            {
                setResolution.Invoke(null, new object[] { previousResolution.x, previousResolution.y });
            }
            yield return host.Localization.ChangeLanguageAsync(previousLanguage).ToCoroutine();
        }

        IEnumerator BoardPress(VisualElement root, string name, Keyboard keyboard)
        {
            Button button = root.Q<Button>(name);
            Assert.That(button, Is.Not.Null, name);
            button.GetFirstAncestorOfType<ScrollView>()?.ScrollTo(button);
            button.Focus(); yield return null;
            Assert.That(button.enabledInHierarchy, Is.True, name);
            _inputFixture.PressAndRelease(keyboard.enterKey); yield return null; yield return null;
        }
